  const savedName = localStorage.getItem("behavior-log-name");

  if (savedName) {

    const chip = document.querySelector("#logger-chips .chip[data-val='" + savedName + "']");

    if (chip) chip.classList.add("active");

    else {

      const otherChip = document.querySelector("#logger-chips .chip[data-val='__other__']");

      if (otherChip) { otherChip.classList.add("active"); }

      const otherInput = document.getElementById("logger-other");

      if (otherInput) { otherInput.value = savedName; otherInput.classList.add("visible"); }

    }

  }


  function initChips(containerId, multiSelect, otherId) {

    document.getElementById(containerId).addEventListener("click", function(e) {

      const chip = e.target.closest(".chip");

      if (!chip) return;

      const isOther = chip.dataset.val === "__other__";


      if (!multiSelect) {

        document.querySelectorAll("#" + containerId + " .chip").forEach(function(c) {

          c.classList.remove("active");

        });

      }

      chip.classList.toggle("active");


      if (otherId) {

        const otherChipActive = document.querySelector("#" + containerId + " .chip[data-val='__other__']").classList.contains("active");

        const el = document.getElementById(otherId);

        el.classList.toggle("visible", otherChipActive);

        if (!otherChipActive) el.value = "";

      }

    });

  }


  initChips("logger-chips",   false, "logger-other");

  initChips("setting-chips",  false, "setting-other");

  initChips("physio-chips",   true,  "physio-other");

  initChips("emotion-chips",  true,  "emotion-other");

  initChips("function-chips", false, "function-other");


  function getChips(containerId, otherId) {

    const vals = Array.from(document.querySelectorAll("#" + containerId + " .chip.active"))

      .filter(function(c) { return c.dataset.val !== "__other__"; })

      .map(function(c) { return c.dataset.val; });


    const otherActive = document.querySelector("#" + containerId + " .chip[data-val='__other__']").classList.contains("active");

    if (otherActive && otherId) {

      const otherVal = document.getElementById(otherId).value.trim();

      if (otherVal) vals.push(otherVal);

    }

    return vals.join(", ");

  }


  async function submitForm() {

    const payload = {

      date:        document.getElementById("f-date").value,

      time:        document.getElementById("f-time").value,

      logger:      getChips("logger-chips", "logger-other"),

      setting:     getChips("setting-chips", "setting-other"),

      physio:      getChips("physio-chips",   "physio-other"),

      antecedent:  document.getElementById("f-antecedent").value.trim(),

      behavior:    document.getElementById("f-behavior").value.trim(),

      consequence: document.getElementById("f-consequence").value.trim(),

      emotion:     getChips("emotion-chips",  "emotion-other"),

      fn:          getChips("function-chips", "function-other"),

      notes:       document.getElementById("f-notes").value.trim(),

      glimmer:     document.getElementById("f-glimmer").checked ? "Yes" : "No"

    };


    const status = document.getElementById("status-msg");


    if (!payload.antecedent || !payload.behavior) {

      status.className = "status err";

      status.textContent = "Please fill in at least the Antecedent and Behavior fields before logging.";

      return;

    }


    if (SCRIPT_URL === "YOUR_APPS_SCRIPT_URL_HERE") {

      status.className = "status err";

      status.textContent = "Apps Script URL not set. Open this file in a text editor, paste your URL, and save.";

      return;

    }


    status.className = "status loading";

    status.textContent = "Saving entry...";


    if (payload.logger) localStorage.setItem("behavior-log-name", payload.logger);


    try {

      await fetch(SCRIPT_URL, {

        method: "POST",

        body: JSON.stringify(payload)

      });

      status.className = "status ok";

      status.textContent = "Entry logged successfully.";

      resetForm();

    } catch (err) {

      status.className = "status err";

      status.textContent = "Could not reach Google Sheets. Check your Script URL and internet connection.";

    }

  }


  function resetForm() {

    ["f-antecedent","f-behavior","f-consequence","f-notes","physio-other","emotion-other","function-other","logger-other","setting-other"].forEach(function(id) {

      var el = document.getElementById(id);

      if (el) el.value = "";

    });

    document.getElementById("f-setting").value = "";

    document.getElementById("f-glimmer").checked = false;

    document.querySelectorAll(".chip.active").forEach(function(c) { c.classList.remove("active"); });

    document.querySelectorAll(".other-input").forEach(function(el) { el.classList.remove("visible"); });


    const now2 = new Date();

    document.getElementById("f-date").value = now2.toISOString().split("T")[0];

    document.getElementById("f-time").value = now2.toTimeString().slice(0, 5);


    const saved = localStorage.getItem("behavior-log-name");

    if (saved) {

      const chip = document.querySelector("#logger-chips .chip[data-val='" + saved + "']");

      if (chip) chip.classList.add("active");

      else {

        const otherChip = document.querySelector("#logger-chips .chip[data-val='__other__']");

        if (otherChip) otherChip.classList.add("active");

        const otherInput = document.getElementById("logger-other");

        if (otherInput) { otherInput.value = saved; otherInput.classList.add("visible"); }

      }

    }
